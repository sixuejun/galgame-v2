## 2. v1 到 v2 的迁移

### 2.1 核心差异回顾

| 方面 | v1 | v2 |
|------|-----|-----|
| 渲染层 | `host: 'div'` | `host: 'iframe'` |
| 数据获取 | 依赖脚本 push | iframe 自主拉取 |
| 楼层浏览 | 不支持 | 支持历史楼层浏览 |
| 块级翻页 | 当前楼层内 | 当前楼层内 + 自动切楼层 |

## 3. Store 拆分指南

### 3.1 为什么要拆分

v1 的 `store.ts` 超过 3000 行，导致：
- 代码难以维护
- 难以定位问题
- 多人协作困难

### 3.2 Store 拆分对照

| v1 Store（全部） | v2 Store | 职责 |
|-----------------|----------|------|
| — | `navigationStore.ts` | 翻页/导航状态（核心） |
| — | `danmakuStore.ts` | 弹幕队列和显示状态 |
| — | `settingsStore.ts` | 用户设置 |
| — | `cardStore.ts` | 卡牌队列 |
| — | `characterStore.ts` | 角色数据 |
| — | `achievementStore.ts` | 成就状态 |
| — | `goldStore.ts` | 金币/经济 |
| — | `inventoryStore.ts` | 背包 |
| — | `apiConfigStore.ts` | 第二 API 配置 |

### 3.3 Store 实现示例

```typescript
// [1] VN舞台/store/navigationStore.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { DialogueUnit, MessageBlock } from '../../types/message';

export const useNavigationStore = defineStore('navigation', () => {
  // === State ===
  const dialogues = ref<DialogueUnit[]>([]);
  const currentDialogueIndex = ref(0);
  const currentBlockIndex = ref(0);
  const previewDialogueIndex = ref(0);
  const duringStreaming = ref(false);

  // === Getters ===
  const currentDialogue = computed(() => dialogues.value[currentDialogueIndex.value]);
  
  const currentBlock = computed(
    () => currentDialogue.value?.blocks[currentBlockIndex.value]
  );

  const isLastBlock = computed(() =>
    currentDialogueIndex.value === dialogues.value.length - 1 &&
    currentBlockIndex.value === currentDialogue.value?.blocks.length - 1
  );

  const isFirstBlock = computed(() =>
    currentDialogueIndex.value === 0 &&
    currentBlockIndex.value === 0
  );

  const totalDialogues = computed(() => dialogues.value.length);

  const currentDialogueBlocks = computed(
    () => currentDialogue.value?.blocks ?? []
  );

  // === Actions ===
  function nextBlock() {
    const blocks = currentDialogue.value?.blocks ?? [];
    
    if (currentBlockIndex.value < blocks.length - 1) {
      currentBlockIndex.value++;
    } else if (currentDialogueIndex.value < dialogues.value.length - 1) {
      currentDialogueIndex.value++;
      currentBlockIndex.value = 0;
    }
  }

  function prevBlock() {
    if (currentBlockIndex.value > 0) {
      currentBlockIndex.value--;
    } else if (currentDialogueIndex.value > 0) {
      currentDialogueIndex.value--;
      const prevBlocks = dialogues.value[currentDialogueIndex.value].blocks;
      currentBlockIndex.value = prevBlocks.length - 1;
    }
  }

  function goToDialogue(index: number) {
    if (index >= 0 && index < dialogues.value.length) {
      currentDialogueIndex.value = index;
      currentBlockIndex.value = 0;
    }
  }

  function appendDialogue(dialogue: DialogueUnit) {
    dialogues.value.push(dialogue);
    currentDialogueIndex.value = dialogues.value.length - 1;
    currentBlockIndex.value = 0;
  }

  return {
    // State
    dialogues,
    currentDialogueIndex,
    currentBlockIndex,
    previewDialogueIndex,
    duringStreaming,
    // Getters
    currentDialogue,
    currentBlock,
    isLastBlock,
    isFirstBlock,
    totalDialogues,
    currentDialogueBlocks,
    // Actions
    nextBlock,
    prevBlock,
    goToDialogue,
    appendDialogue,
  };
});
```

### 4.2 触摸互动系统

#### 4.2.1 区域定义

在世界书 JSON 中定义触摸区域（见 4.1.1）。

#### 4.2.2 触摸检测实现

```typescript
// composables/useTouchInteraction.ts
export function useTouchInteraction(
  canvasRef: Ref<HTMLCanvasElement | null>,
  touchAreas: TouchArea[]
) {
  const pressedArea = ref<string | null>(null);
  const pressStartTime = ref<number>(0);
  const currentPosition = ref({ x: 0, y: 0 });

  function getHitArea(x: number, y: number): TouchArea | null {
    const canvas = canvasRef.value;
    if (!canvas) return null;

    const rect = canvas.getBoundingClientRect();
    const relX = (x - rect.left) / rect.width;
    const relY = (y - rect.top) / rect.height;

    for (const area of touchAreas) {
      if (
        relX >= area.hitbox.x &&
        relX <= area.hitbox.x + area.hitbox.width &&
        relY >= area.hitbox.y &&
        relY <= area.hitbox.y + area.hitbox.height
      ) {
        return area;
      }
    }
    return null;
  }

  function handlePointerDown(e: PointerEvent) {
    const area = getHitArea(e.clientX, e.clientY);
    if (area) {
      pressedArea.value = area.id;
      pressStartTime.value = Date.now();
      currentPosition.value = { x: e.clientX, y: e.clientY };
      startEffect(area, e.clientX, e.clientY);
    }
  }

  function handlePointerUp(e: PointerEvent) {
    if (!pressedArea.value) return;
    
    const area = touchAreas.find(a => a.id === pressedArea.value);
    const duration = Date.now() - pressStartTime.value;
    const threshold = area?.shortPressThreshold ?? 400;

    if (duration < threshold) {
      triggerShortPress(area);
    } else {
      triggerLongPress(area);
    }

    pressedArea.value = null;
    stopEffect();
  }

  return {
    pressedArea,
    currentPosition,
    handlePointerDown,
    handlePointerUp,
  };
}
```

### 4.3 第二 API 统一调用层

```typescript
// 共享/utils/api/secondApiClient.ts
interface SecondApiConfig {
  enabled: boolean;
  baseUrl: string;
  apiKey: string;
  model: string;
  temperature?: number;
  maxTokens?: number;
}

export async function callSecondApi(
  task: 'danmaku' | 'imageTag' | 'shop' | 'riddle' | 'comms' | 'boardGameEvent',
  payload: {
    contentText?: string;
    keyword?: string;
    lastMessage?: string;
    chatHistory?: string;
  }
): Promise<string | string[]> {
  const config = useApiConfigStore().config;
  if (!config.enabled) throw new Error('Second API is disabled');

  const systemPrompt = buildSystemPromptForTask(task, payload);
  const worldbookPrompts = worldbookLoader.getSecondApiPrompts();

  const response = await fetch(`${config.baseUrl}/chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(config.apiKey ? { Authorization: `Bearer ${config.apiKey}` } : {}),
    },
    body: JSON.stringify({
      model: config.model,
      messages: [
        { role: 'system', content: [...worldbookPrompts, systemPrompt].join('\n\n') },
        { role: 'user', content: payload.contentText || payload.lastMessage || ' ' },
      ],
      temperature: config.temperature ?? 0.7,
    }),
  });

  if (!response.ok) throw new Error(`API error: ${response.status}`);
  
  const data = await response.json();
  return data.choices[0].message.content;
}
```

### 4.4 足迹系统

```typescript
// [2]世界书联动/utils/footprintParser.ts
interface Footprint {
  icon: string;
  title: string;
  condition: string;
  reward: string;
  evaluation: string;
}

const FOOTPRINT_REGEX = /\*{0,2}【\s*(?:触发)?成就[^：:|｜]*?[：:|｜]\s*[（(]?\s*([^\s)）|｜】]{1,3})\s*[)）]?\s*([^|｜】]*?)\s*[|｜]\s*(?:触发)?(?:条件)?[：:]?\s*([\s\S]*?)\s*[|｜]\s*(?:奖励)?[：:]?\s*([\s\S]*?)(?:\s*[|｜]\s*(?:系统)?(?:评价)?[：:]?\s*([\s\S]*?))?\s*】\*{0,2}/gu;

export function parseFootprint(text: string): Footprint | null {
  const match = FOOTPRINT_REGEX.exec(text);
  if (!match) return null;

  return {
    icon: match[1]?.trim() || '📌',
    title: match[2]?.trim() || '',
    condition: match[4]?.trim() || '',
    reward: match[5]?.trim() || '',
    evaluation: match[6]?.trim() || '',
  };
}
```

3. 在 `pluginStore.ts` 中注册

```typescript
// pluginStore.ts
export const usePluginStore = defineStore('plugins', () => {
  const plugins = ref([
    {
      id: 'my-plugin',
      name: '我的插件',
      icon: 'fa-star',
      component: () => import('./components/MyPlugin.vue'),
    },
    // ...
  ]);

  return { plugins };
});
```

### Q3: 如何处理跨模块通信？

使用 Pinia 的响应式特性：

```typescript
// 在 Store A 中
const storeA = useStoreA();
watch(() => storeA.someValue, (newVal) => {
  const storeB = useStoreB();
  storeB.reactToStoreA(newVal);
});
```

### Q4: 如何持久化数据？

两种方式，根据数据用途选择：

```typescript
// 方式 1：酒馆变量（需要 AI 感知）
watchEffect(() => {
  replaceVariables({ key: value }, { type: 'chat' });
});

// 方式 2：localStorage（纯前端）
localStorage.setItem('my-data', JSON.stringify(data));
```
